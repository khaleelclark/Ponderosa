cmake_minimum_required(VERSION 3.20)
project(ponderosa_speech LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SHERPA_ROOT "" CACHE PATH "Verified sherpa-onnx 1.13.8 distribution")
if(NOT EXISTS "${SHERPA_ROOT}/include/sherpa-onnx/c-api/c-api.h")
  message(FATAL_ERROR "Provide the verified pinned sherpa-onnx SDK via SHERPA_ROOT")
endif()
add_executable(ponderosa-speech worker.cc)
target_include_directories(ponderosa-speech PRIVATE "${SHERPA_ROOT}/include")
find_library(SHERPA_API NAMES sherpa-onnx-c-api PATHS "${SHERPA_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(ponderosa-speech PRIVATE "${SHERPA_API}")
if(APPLE)
  set_target_properties(ponderosa-speech PROPERTIES INSTALL_RPATH "@loader_path/lib" BUILD_WITH_INSTALL_RPATH TRUE)
elseif(UNIX)
  set_target_properties(ponderosa-speech PROPERTIES INSTALL_RPATH "$ORIGIN/lib" BUILD_WITH_INSTALL_RPATH TRUE)
endif()
install(TARGETS ponderosa-speech RUNTIME DESTINATION .)
if(WIN32)
  install(DIRECTORY "${SHERPA_ROOT}/lib/" DESTINATION . FILES_MATCHING PATTERN "*.dll")
  install(DIRECTORY "${SHERPA_ROOT}/bin/" DESTINATION . OPTIONAL FILES_MATCHING PATTERN "*.dll")
else()
  install(DIRECTORY "${SHERPA_ROOT}/lib/" DESTINATION lib FILES_MATCHING PATTERN "*.so*" PATTERN "*.dylib")
endif()
